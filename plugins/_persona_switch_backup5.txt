_NS = "persona_switch"
_SWITCH_DELAY = 4.0

PLUGIN = {
    "name": "persona_switch",
    "description": (
        "Switches the assistant's persona: name, voice and personality. Each switch "
        "restarts the conversation, so call it ONLY when the user clearly asks to "
        "change persona or voice. The user just saying a name ('Jarvis', 'Riya', "
        "'Max') to get your attention is NOT a request to switch. Use when "
        "the user says things like 'switch to Jarvis', 'Riya ko bulao', 'Max pe "
        "switch karo', 'male voice karo', 'female voice karo', or asks which "
        "personas exist. Available personas: "
        + ", ".join(PERSONAS) +
        ". male voice -> jarvis, female voice -> riya."
    ),
    "parameters": {
        "type": "OBJECT",
        "properties": {
            "persona": {
                "type": "STRING",
                "description": " | ".join(PERSONAS) + " | list",
            },
        },
        "required": ["persona"],
    },
}


def run(parameters: dict, player=None, session_memory=None) -> str:
    try:
        key = str((parameters or {}).get("persona", "")).strip().lower()
        if key in ("male", "boy", "aadmi"):
            key = "jarvis"
        if key in ("female", "girl", "girlfriend", "ladki"):
            key = "riya"
        if key in ("", "list"):
            return "status: personas available: " + ", ".join(
                f"{k} ({v['name']}, voice {v['voice']})" for k, v in PERSONAS.items())
        p = PERSONAS.get(key)
        if p is None:
            return ("failed: no persona called '" + key + "'. Available: "
                    + ", ".join(PERSONAS) + ".")

        from memory.config_manager import (save_assistant_config, get_user_name,
                                           save_voice, save_plugin_config,
                                           get_plugin_setting, get_voice)
        already_active = (get_plugin_setting(_NS, "current", "") == key
                          and get_voice() == p["voice"])
        if already_active:
            return (f"status: {p['name']} is ALREADY the active persona. Nothing was "
                    f"changed and no restart is needed. Do not call this tool again "
                    f"unless the user clearly asks for a DIFFERENT persona.")
        save_assistant_config(p["name"], get_user_name())
        save_voice(p["voice"])
        save_plugin_config(_NS, {"prompt": p["prompt"], "current": key})

        rebuild = getattr(player, "on_voice_change", None)
        if callable(rebuild):
            threading.Timer(_SWITCH_DELAY, rebuild).start()
            try:
                player.write_log(f"SYS: Persona -> {p['name']} ({p['voice']})")
            except Exception:
                pass
            return (f"done: switching to {p['name']}. Say ONE short line in your "
                    f"current voice now; in a few seconds the session restarts as "
                    f"{p['name']} with the new voice.")
        return (f"unverified: saved {p['name']} as the persona, but could not "
                f"restart the session. It applies on the next launch.")
    except Exception as e:
        return f"failed: could not switch persona: {e}"
